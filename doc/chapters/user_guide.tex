\chapter{راهنمای کاربر}\label{Chap:UserGuide}

این فصل روش استفاده از برنامهٔ «نوواژه» را از دید والد یا سرپرست کودک توضیح می‌دهد. برنامه شامل آموزش واژه، آزمون شنیداری، بازی‌های شناختی، محتوای شخصی و حالت کودک است. بهتر است والد در نخستین استفاده تنظیمات را انجام دهد و هنگام فعالیت، کودک را همراهی کند. نوواژه یک ابزار آموزشی مکمل است و نتیجهٔ بازی‌ها جای ارزیابی گفتاردرمانگر یا پزشک را نمی‌گیرد.

\section{شروع سریع}

برای استفادهٔ سریع مراحل زیر انجام می‌شوند:

\begin{enumerate}
\item برنامه را باز کنید و در صورت نیاز حساب والد بسازید.
\item از زبانهٔ «تنظیمات»، یک نمایه برای کودک اضافه یا کودک مورد نظر را انتخاب کنید.
\item زمان مجاز فعالیت را روی مقدار کوتاهی مانند ۵ یا ۱۰ دقیقه قرار دهید.
\item به «داشبورد» برگردید و یک دستهٔ آموزشی یا یکی از بازی‌ها را انتخاب کنید.
\item کنار کودک بمانید، نام تصویرها را تکرار کنید و آن‌ها را به اشیای واقعی مرتبط سازید.
\item اگر تلفن را به کودک می‌سپارید، «حالت کودک» را فعال کنید تا خروج ناخواسته از برنامه دشوار شود.
\end{enumerate}

بازی‌های عمومی بدون ورود نیز در دسترس‌اند، اما برای ساخت نمایهٔ کودک، افزودن واژهٔ شخصی و ثبت روند یادگیری باید وارد حساب والد شد.

\section{ساخت حساب و ورود}

در صفحهٔ ورود، «نام کاربری» و «رمز عبور» را وارد و دکمهٔ «ورود» را لمس کنید. اگر هنوز حسابی ندارید، پیوند «حساب کاربری ندارید؟ ثبت‌نام کنید» را انتخاب نمایید. در صفحهٔ ثبت‌نام سه مقدار نام کاربری، نام نمایشی و رمز عبور دریافت می‌شوند. نام نمایشی همان نامی است که برنامه برای خطاب‌کردن والد نمایش می‌دهد.

\section{داشبورد و تنظیمات والد}

پس از ورود، صفحهٔ اصلی یا «داشبورد» نمایش داده می‌شود. در بالای صفحه نام کودک فعال قرار دارد. کارت‌های رنگی بالای صفحه مسیر بازی‌ها و آزمون هستند و بخش «آموزش» دسته‌هایی مانند میوه‌ها، سبزیجات، حیوانات و غذاها را نشان می‌دهد. با لمس هر دسته، آموزش واژه‌های همان گروه آغاز می‌شود.

نوار پایین صفحه دو بخش اصلی دارد: «داشبورد» برای فعالیت کودک و «تنظیمات» برای مدیریت حساب، فرزندان، واژه‌ها، زمان و حالت کودک. بخش تنظیمات باید توسط والد استفاده شود.

\subsection{افزودن و انتخاب کودک}

در نخستین ورود، اگر هنوز نمایه‌ای وجود نداشته باشد، فرم افزودن فرزند نمایش داده می‌شود. نام، سن و جنسیت را انتخاب و «افزودن فرزند» را بزنید. برنامه حداکثر دو نمایهٔ کودک را پشتیبانی می‌کند. برای تغییر کودک فعال، وارد «تنظیمات» شوید و نام کودک مورد نظر را لمس کنید؛ برنامه به داشبورد برمی‌گردد و نام همان کودک در بالای صفحه نمایش داده می‌شود.

ثبت پاسخ و پیشرفت به کودک فعال وابسته است. پیش از شروع آموزش یا آزمون، نام بالای داشبورد را کنترل کنید تا نتیجه برای نمایهٔ درست ذخیره شود. اطلاعات کودک باید فقط در حساب سرپرست نگهداری شود و بهتر است از واردکردن اطلاعات غیرضروری خودداری گردد.

\section{آموزش واژه‌ها}

در بخش «آموزش»، دستهٔ دلخواه را لمس کنید. برنامه واژه‌های آن دسته را یکی‌یکی با تصویر و نام فارسی نمایش می‌دهد. دکمهٔ پخش، تلفظ واژه را پخش می‌کند و پیکان‌های دو طرف برای رفتن به واژهٔ قبلی یا بعدی هستند. نوار کوچک زیر واژه میزان پیشرفت همان واژه را نشان می‌دهد و شمارندهٔ بالای صفحه، تعداد واژه‌های آموخته‌شده در دسته را نمایش می‌دهد.

برای استفادهٔ مؤثر، اجازه دهید کودک ابتدا تصویر را ببیند و صدا را بشنود. سپس نام را یک یا دو بار طبیعی تکرار کنید و برای پاسخ مکث نمایید. از مجبورکردن کودک به تکرار پی‌درپی خودداری کنید. بهتر است پس از هر چند واژه، نمونهٔ واقعی یا اسباب‌بازی مرتبط را نیز نشان دهید.

\section{آزمون شنیداری}

برای شروع آزمون، کارت «آزمون» را در داشبورد لمس کنید. سپس یک دسته یا حالت ترکیبی را انتخاب نمایید. در هر پرسش، دکمهٔ پخش را بزنید، به واژه گوش کنید و تصویر درست را انتخاب نمایید. نوار بالای صفحه شمارهٔ پرسش و میزان پیشرفت جلسه را نشان می‌دهد. هر جلسهٔ عادی شامل پنج پرسش است.

اگر کودک پاسخ اشتباه داد، ابتدا دوباره صدا را پخش کنید و فرصت تلاش بدهید. پاسخ را با سرزنش یا عجله همراه نکنید. در پایان، نتیجهٔ جلسه نمایش داده می‌شود و در صورت فعال‌بودن نمایهٔ کودک، پاسخ‌ها در روند او ثبت می‌شوند. عددهای برنامه نشان‌دهندهٔ عملکرد در همین فعالیت‌اند و نباید به‌عنوان تشخیص رشد زبان تفسیر شوند.

\section{بازی‌های کودک}

کارت‌های بازی در بالای داشبورد قرار دارند. بعضی بازی‌ها پیش از شروع اجازه می‌دهند یک دسته یا حالت ترکیبی انتخاب شود. دکمهٔ خانه در هر بازی برای بازگشت به داشبورد و «شروع دوباره» برای ساخت یک دور تازه است.

\subsection{بازی حافظه و بازی رنگ‌ها}

در «بازی حافظه» کارت‌ها را دو‌به‌دو لمس کنید تا جفت‌های مشابه پیدا شوند. بهتر است والد نام تصویر هر کارت را هنگام بازشدن بیان کند. تعداد ستاره‌ها نتیجهٔ دور را نشان می‌دهد و پس از پایان می‌توان بازی را دوباره آغاز کرد.

در «بازی رنگ‌ها» کودک بادکنک‌ها را نگه می‌دارد و به جعبهٔ همرنگ می‌کشد. این بازی نام رنگ‌ها، توجه دیداری و هماهنگی لمس و حرکت را تمرین می‌دهد. اگر کشیدن برای کودک دشوار است، والد می‌تواند دست او را راهنمایی کند، ولی بهتر است فرصت تلاش مستقل نیز داده شود.

\section{افزودن دسته و واژهٔ شخصی}

محتوای شخصی به والد اجازه می‌دهد واژه‌های آشنای زندگی کودک، مانند نام اعضای خانواده یا اسباب‌بازی محبوب، را به برنامه اضافه کند. برای ورود به این بخش، «تنظیمات» و سپس «افزودن کلمه» را انتخاب نمایید.

ابتدا فهرست دسته‌ها نمایش داده می‌شود. می‌توان یکی از دسته‌های موجود را انتخاب کرد یا «دسته‌بندی جدید» را زد. برای دستهٔ جدید، نام فارسی الزامی است؛ نام انگلیسی و عکس دسته اختیاری‌اند. بهتر است نام دسته کوتاه و تصویر آن روشن و مرتبط باشد.

\section{تنظیم زمان و حالت کودک}

در زبانهٔ «تنظیمات» می‌توان زمان مجاز فعالیت را روی ۱، ۵، ۱۰، ۱۵ یا ۳۰ دقیقه قرار داد. برای کودک خردسال بهتر است از جلسه‌های کوتاه شروع شود. پایان زمان به معنای پایان تعامل نیست؛ والد می‌تواند واژه‌های همان جلسه را بدون صفحه و در بازی روزمره ادامه دهد.

«حالت کودک» از خروج ناخواسته از برنامه جلوگیری می‌کند. با لمس این گزینه، پنجره‌ای شامل «قفل کن»، «تغییر رمز» و راهنمای فعال‌سازی باز می‌شود. در نخستین استفاده یک رمز چهاررقمی مخصوص والد تعیین کنید. این رمز نباید در اختیار کودک قرار گیرد.

اگر پیام قرمز فعال‌نبودن پین نمایش داده شد، راهنمای بازشده را دنبال و قابلیت را در تنظیمات تلفن روشن کنید. حالت کودک یک کمک فنی است و جای نظارت والد را نمی‌گیرد. روش قفل در آی‌اواس یا نسخه‌های مختلف سیستم‌عامل ممکن است متفاوت باشد و باید مطابق راهنمای همان دستگاه انجام شود.

\section{رفع مشکلات رایج}

\begin{itemize}
\item \textbf{تصویر یا صدا بارگذاری نمی‌شود:} اتصال اینترنت را بررسی کنید، چند لحظه منتظر بمانید و دوباره وارد صفحه شوید.
\item \textbf{صدای واژه پخش نمی‌شود:} صدای رسانهٔ تلفن و حالت بی‌صدا را بررسی و دکمهٔ پخش را دوباره لمس کنید.
\item \textbf{ضبط صدا یا انتخاب تصویر کار نمی‌کند:} مجوز میکروفن، دوربین و دسترسی به تصویرها را در تنظیمات تلفن بررسی کنید.
\item \textbf{پیشرفت ذخیره نمی‌شود:} مطمئن شوید وارد حساب والد شده‌اید و کودک درست در تنظیمات انتخاب شده است.
\item \textbf{دسته یا واژه دیده نمی‌شود:} صفحه را دوباره باز کنید و از ثبت موفق محتوا و اتصال شبکه مطمئن شوید.
\item \textbf{حالت کودک فعال نمی‌شود:} ابتدا پین‌کردن برنامه را در تنظیمات اندروید فعال و سپس از داخل نوواژه گزینهٔ «قفل کن» را انتخاب کنید.
\item \textbf{برنامه پاسخ نمی‌دهد:} برنامه را ببندید و دوباره باز کنید. اگر مشکل ادامه داشت، اتصال شبکه و نسخهٔ برنامه را بررسی نمایید.
\end{itemize}

\section{توصیه‌های استفادهٔ مناسب}

برای کودکان زیر سه سال، جلسهٔ کوتاه و همراهی والد مهم‌تر از تعداد مرحله‌های تکمیل‌شده است. فعالیت را هنگامی شروع کنید که کودک خسته یا بی‌قرار نیست. هر واژه را به یک حرکت، شیء واقعی یا گفت‌وگوی کوتاه پیوند دهید و در صورت بی‌علاقگی کودک، جلسه را پایان دهید. پیشرفت را بر اساس تعامل روزمره مشاهده کنید، نه فقط امتیاز برنامه.

نوواژه برای تمرین و ایجاد فرصت گفت‌وگو طراحی شده است. اگر دربارهٔ شنوایی، درک زبان، استفاده از اشاره، تولید واژه یا از‌دست‌رفتن مهارت‌های کودک نگرانی وجود دارد، با پزشک یا آسیب‌شناس گفتار و زبان مشورت کنید.
